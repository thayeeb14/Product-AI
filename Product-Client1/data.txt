package com.iss.Product_Client1.controllers;

import com.iss.Product_Client1.services.ClientService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.*;

@Controller
@RequestMapping("/ai")
public class ClientController {

    @Autowired
    ClientService clientService;

    private final String conversationId = "shopping-chat";

    @GetMapping("")
    public String home(Model model) {

        model.addAttribute("conversationId", conversationId);
        model.addAttribute("history",
                clientService.getChatHistory(conversationId));

        return "aihome";
    }

    @PostMapping("/chat")
    public String chat(
            @RequestParam String prompt,
            Model model) {

        String response = clientService.chat(prompt, conversationId);

        model.addAttribute("prompt", "");
        model.addAttribute("response", response);
        model.addAttribute("conversationId", conversationId);

        model.addAttribute("history",
                clientService.getChatHistory(conversationId));

        return "aihome";
    }
}
